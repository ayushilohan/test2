// read json in tizen 
function readFeatureConfig(callback) {
  tizen.filesystem.resolve(
    'wgt-private/feature-config.json', // adjust path if placed elsewhere
    function(dir) {
      dir.readAsText(function(content) {
        try {
          const config = JSON.parse(content);
          console.log("✅ Feature Config Loaded:", config);
          callback(config);
        } catch (e) {
          console.error("❌ Error parsing config JSON", e);
        }
      }, function(err) {
        console.error("❌ Error reading file:", err);
      });
    },
    function(err) {
      console.error("❌ Failed to resolve file path:", err);
    },
    'r' // read-only
  );
}
