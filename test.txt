var currentFocus = 0;
var totalItems = 2;
var contentDisplay = document.getElementById('content-display');

function loadPage(index) {
  var page = '';
  if (index === 0) page = 'home.html';
  else if (index === 1) page = 'settings.html';

  fetch('pages/' + page)
    .then(function(res) {
      return res.text();
    })
    .then(function(html) {
      contentDisplay.innerHTML = html;
    })
    .catch(function() {
      contentDisplay.innerHTML = '<p>Error loading page.</p>';
    });
}

function updateFocus() {
  for (var i = 0; i < totalItems; i++) {
    var item = document.getElementById('menu-' + i);
    if (i === currentFocus) {
      item.classList.add('focused');
    } else {
      item.classList.remove('focused');
    }
  }
  loadPage(currentFocus);
}

document.addEventListener('keydown', function(e) {
  if (e.key === 'ArrowDown') {
    currentFocus = (currentFocus + 1) % totalItems;
    updateFocus();
  } else if (e.key === 'ArrowUp') {
    currentFocus = (currentFocus - 1 + totalItems) % totalItems;
    updateFocus();
  }
});

updateFocus(); // Load initial content
