const items = document.querySelectorAll('.menu-item');
const contentDisplay = document.getElementById('content-display');
let currentFocus = 0;

const contentMap = {
  0: `<h2>🏠 Home</h2><p>Welcome to the home screen.</p>`,
  1: `<h2>⚙️ Settings</h2><p>Adjust your preferences here.</p>`
};

function updateFocus() {
  items.forEach((item, index) => {
    item.classList.toggle('focused', index === currentFocus);
  });
  contentDisplay.innerHTML = contentMap[currentFocus];
}

document.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowDown') {
    currentFocus = (currentFocus + 1) % items.length;
    updateFocus();
  } else if (e.key === 'ArrowUp') {
    currentFocus = (currentFocus - 1 + items.length) % items.length;
    updateFocus();
  }
});

updateFocus(); // Initial render
